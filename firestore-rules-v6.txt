rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Posts: anyone can read; only the signed-in owner can create;
    // owner (or the site admin) can update/delete.
    match /posts/{postId} {
      allow read: if true;
      allow create: if request.auth != null
                    && request.auth.uid == request.resource.data.owner;
      allow update, delete: if request.auth != null
                    && (request.auth.uid == resource.data.owner
                        || request.auth.token.email == 'mimpakhi05@gmail.com');
    }

    // Users: profiles are public to read; each user writes only their own doc.
    match /users/{userId} {
      allow read: if true;
      allow create, update: if request.auth != null && request.auth.uid == userId;
      allow delete: if false;
    }

    // Exchange requests: only the two people involved can read/update;
    // a request can only be created in your own name; never deleted.
    match /requests/{reqId} {
      allow read: if request.auth != null
                  && (request.auth.uid == resource.data.fromUid
                      || request.auth.uid == resource.data.toUid);
      allow create: if request.auth != null
                    && request.auth.uid == request.resource.data.fromUid;
      allow update: if request.auth != null
                    && (request.auth.uid == resource.data.fromUid
                        || request.auth.uid == resource.data.toUid);
      allow delete: if false;
    }

    // Ratings: public to read; you can only rate as yourself; never edited.
    match /ratings/{ratingId} {
      allow read: if true;
      allow create: if request.auth != null
                    && request.auth.uid == request.resource.data.fromUid;
      allow update, delete: if false;
    }

    // Points ledger (Phase H): you can read your own entries; an entry is
    // created by the signed-in user who performed the action; never edited.
    match /points/{docId} {
      allow read: if request.auth != null && request.auth.uid == resource.data.uid;
      allow create: if request.auth != null
                    && request.auth.uid == request.resource.data.by;
      allow update, delete: if false;
    }

    // File unlocks bought with points (Phase H): your own records only.
    match /unlocks/{docId} {
      allow read: if request.auth != null && request.auth.uid == resource.data.uid;
      allow create: if request.auth != null
                    && request.auth.uid == request.resource.data.uid;
      allow update, delete: if false;
    }

    // Claimable rewards (Phase K): you read your own rewards; a reward is
    // created by the signed-in user who performed the earning action;
    // only the owner can update it (to mark it claimed); never deleted.
    match /rewards/{docId} {
      allow read: if request.auth != null && request.auth.uid == resource.data.uid;
      allow create: if request.auth != null
                    && request.auth.uid == request.resource.data.by;
      allow update: if request.auth != null && request.auth.uid == resource.data.uid;
      allow delete: if false;
    }

    // Community wall (Phase J): public to read; post only as yourself;
    // delete only your own post (or the site admin).
    match /community/{docId} {
      allow read: if true;
      allow create: if request.auth != null
                    && request.auth.uid == request.resource.data.fromUid;
      allow update: if false;
      allow delete: if request.auth != null
                    && (request.auth.uid == resource.data.fromUid
                        || request.auth.token.email == 'mimpakhi05@gmail.com');
    }

    // Private messages (Phase J): only the two people in the chat can
    // read/update (update is for read receipts); send only as yourself.
    match /messages/{docId} {
      allow read: if request.auth != null
                  && (request.auth.uid == resource.data.fromUid
                      || request.auth.uid == resource.data.toUid);
      allow create: if request.auth != null
                    && request.auth.uid == request.resource.data.fromUid;
      allow update: if request.auth != null
                    && (request.auth.uid == resource.data.fromUid
                        || request.auth.uid == resource.data.toUid);
      allow delete: if false;
    }
  }
}
